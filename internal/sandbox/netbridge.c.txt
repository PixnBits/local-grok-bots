#define _GNU_SOURCE
#include <arpa/inet.h>
#include <errno.h>
#include <netinet/in.h>
#include <signal.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/socket.h>
#include <sys/time.h>
#include <sys/un.h>
#include <unistd.h>

/* Listen on 127.0.0.1 inside the jail and splice each TCP connection to the
   daemon HTTP egress socket. This program does not decide what is allowed. */

static int dial_unix(const char *path) {
    int fd;
    struct sockaddr_un un;
    if (!path || !path[0]) return -1;
    fd = socket(AF_UNIX, SOCK_STREAM, 0);
    if (fd < 0) return -1;
    memset(&un, 0, sizeof un);
    un.sun_family = AF_UNIX;
    if (strlen(path) >= sizeof un.sun_path) {
        close(fd);
        return -1;
    }
    memcpy(un.sun_path, path, strlen(path) + 1);
    if (connect(fd, (struct sockaddr *)&un, sizeof un) < 0) {
        close(fd);
        return -1;
    }
    return fd;
}

static void copy_one(int from, int to) {
    char buf[8192];
    for (;;) {
        ssize_t n = read(from, buf, sizeof buf);
        if (n == 0) {
            shutdown(to, SHUT_WR);
            return;
        }
        if (n < 0) {
            if (errno == EINTR) continue;
            shutdown(from, SHUT_RDWR);
            shutdown(to, SHUT_RDWR);
            return;
        }
        ssize_t off = 0;
        while (off < n) {
            ssize_t w = write(to, buf + off, (size_t)(n - off));
            if (w < 0) {
                if (errno == EINTR) continue;
                shutdown(from, SHUT_RDWR);
                shutdown(to, SHUT_RDWR);
                return;
            }
            if (w == 0) {
                shutdown(from, SHUT_RDWR);
                shutdown(to, SHUT_RDWR);
                return;
            }
            off += w;
        }
    }
}

static int serve(int port, const char *sock) {
    int ln, one = 1;
    struct sockaddr_in in;
    ln = socket(AF_INET, SOCK_STREAM, 0);
    if (ln < 0) return 1;
    setsockopt(ln, SOL_SOCKET, SO_REUSEADDR, &one, sizeof one);
    memset(&in, 0, sizeof in);
    in.sin_family = AF_INET;
    in.sin_addr.s_addr = htonl(0x7f000001);
    in.sin_port = htons((unsigned short)port);
    if (bind(ln, (struct sockaddr *)&in, sizeof in) < 0) return 1;
    if (listen(ln, 16) < 0) return 1;
    /* Line-buffered stdout is a file when the wrapper redirects it. Flush
       before accept so the wrapper can see that listen has returned. */
    setvbuf(stdout, NULL, _IONBF, 0);
    printf("ready\n");
    fflush(stdout);
    signal(SIGCHLD, SIG_IGN);
    signal(SIGPIPE, SIG_IGN);
    for (;;) {
        int c = accept(ln, 0, 0);
        if (c < 0) {
            if (errno == EINTR) continue;
            return 1;
        }
        if (fork() == 0) {
            int u;
            close(ln);
            u = dial_unix(sock);
            if (u < 0) _exit(0);
            if (fork() == 0) {
                copy_one(c, u);
                _exit(0);
            }
            copy_one(u, c);
            close(u);
            close(c);
            _exit(0);
        }
        close(c);
    }
}

static int probe(int port) {
    int fd, rc;
    struct sockaddr_in in;
    struct timeval tv;
    fd = socket(AF_INET, SOCK_STREAM, 0);
    if (fd < 0) return 1;
    memset(&tv, 0, sizeof tv);
    tv.tv_sec = 1;
    setsockopt(fd, SOL_SOCKET, SO_SNDTIMEO, &tv, sizeof tv);
    memset(&in, 0, sizeof in);
    in.sin_family = AF_INET;
    in.sin_addr.s_addr = htonl(0x7f000001);
    in.sin_port = htons((unsigned short)port);
    rc = connect(fd, (struct sockaddr *)&in, sizeof in);
    close(fd);
    return rc == 0 ? 0 : 1;
}

int main(int argc, char **argv) {
    int port;
    /* Own process group so the wrapper can kill the listener and the
       forked splices together when the jail command exits. */
    if (setsid() < 0) {
        /* A process-group leader cannot setsid. Keep going. */
    }
    if (argc == 3 && strcmp(argv[1], "probe") == 0) {
        port = atoi(argv[2]);
        if (port < 1 || port > 65535) return 2;
        return probe(port);
    }
    if (argc != 3) return 2;
    port = atoi(argv[1]);
    if (port < 1 || port > 65535 || !argv[2] || !argv[2][0]) return 2;
    return serve(port, argv[2]);
}
