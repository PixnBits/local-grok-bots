/* Redirect non-loopback TCP connect() to a daemon proxy over a unix socket.
   The proxy is outside the network namespace and enforces the allowlist.
   Names are mapped to 240.0.0.0/8 so getaddrinfo does not need DNS in the jail. */

static int (*real_connect)(int, const struct sockaddr *, socklen_t);
static int (*real_getaddrinfo)(const char *, const char *, const struct addrinfo *, struct addrinfo **);
static pthread_once_t once = PTHREAD_ONCE_INIT;
static pthread_mutex_t mu = PTHREAD_MUTEX_INITIALIZER;
static char hosts[256][254];
static int hostn;

static void init_syms(void) {
    real_connect = dlsym(RTLD_NEXT, "connect");
    real_getaddrinfo = dlsym(RTLD_NEXT, "getaddrinfo");
}

static void lower_copy(char *dst, size_t n, const char *src) {
    size_t i = 0;
    if (n == 0) return;
    for (; src[i] && i + 1 < n; i++) {
        char c = src[i];
        if (c >= 'A' && c <= 'Z') c = (char)(c - 'A' + 'a');
        dst[i] = c;
    }
    dst[i] = 0;
}

/* The daemon dial is bounded at 10s. Wait a little longer so a DENY is
   still read, then fail the handshake instead of blocking the caller. */
#define HANDSHAKE_MS 12000

struct proxy_job {
    int client;
    int pad;
    int port;
    char path[108];
    char host[256];
};

struct copy_arg {
    int from;
    int to;
};

static long mono_ms(void) {
    struct timespec ts;
    if (clock_gettime(CLOCK_MONOTONIC, &ts) != 0) return 0;
    return (long)ts.tv_sec * 1000L + (long)ts.tv_nsec / 1000000L;
}

static int poll_until(int fd, short ev, long deadline) {
    for (;;) {
        long left = deadline - mono_ms();
        struct pollfd pfd;
        int r;
        if (left <= 0) {
            errno = ETIMEDOUT;
            return -1;
        }
        if (left > 10000) left = 10000;
        pfd.fd = fd;
        pfd.events = ev;
        pfd.revents = 0;
        r = poll(&pfd, 1, (int)left);
        if (r < 0) {
            if (errno == EINTR) continue;
            return -1;
        }
        if (r == 0) continue;
        if (pfd.revents & (POLLERR | POLLNVAL)) {
            errno = EIO;
            return -1;
        }
        return 0;
    }
}

static int apply_status(int fd, int fl) {
    int set = 0;
    if (fl & O_NONBLOCK) set |= O_NONBLOCK;
    if (fl & O_APPEND) set |= O_APPEND;
#ifdef O_ASYNC
    if (fl & O_ASYNC) set |= O_ASYNC;
#endif
    if (fcntl(fd, F_SETFL, set) < 0) return -1;
    return 0;
}

static int apply_fdflags(int fd, int fdfl) {
    if (fcntl(fd, F_SETFD, fdfl & FD_CLOEXEC) < 0) return -1;
    return 0;
}

static int dial_unix_nb(const char *path, long deadline) {
    int u, fl, so;
    socklen_t sl;
    struct sockaddr_un un;
    size_t n;

    u = socket(AF_UNIX, SOCK_STREAM, 0);
    if (u < 0) return -1;
    fl = fcntl(u, F_GETFL);
    if (fl < 0 || fcntl(u, F_SETFL, fl | O_NONBLOCK) < 0 || fcntl(u, F_SETFD, FD_CLOEXEC) < 0) {
        close(u);
        return -1;
    }
    memset(&un, 0, sizeof un);
    un.sun_family = AF_UNIX;
    n = strlen(path);
    if (n == 0 || n >= sizeof un.sun_path) {
        close(u);
        errno = ENAMETOOLONG;
        return -1;
    }
    memcpy(un.sun_path, path, n + 1);
    if (real_connect(u, (struct sockaddr *)&un, sizeof un) < 0) {
        if (errno != EINPROGRESS && errno != EAGAIN) {
            int e = errno;
            close(u);
            errno = e;
            return -1;
        }
        if (poll_until(u, POLLOUT, deadline) < 0) {
            close(u);
            return -1;
        }
        so = 0;
        sl = sizeof so;
        if (getsockopt(u, SOL_SOCKET, SO_ERROR, &so, &sl) < 0 || so != 0) {
            if (so == 0) so = errno;
            close(u);
            errno = so ? so : ECONNREFUSED;
            return -1;
        }
    }
    return u;
}

static int proxy_handshake(int u, const char *host, int port, long deadline) {
    char line[400];
    char resp[80];
    int n, off, rn;

    n = snprintf(line, sizeof line, "CONNECT %s %d\n", host, port);
    if (n <= 0 || n >= (int)sizeof line) {
        errno = EINVAL;
        return -1;
    }
    off = 0;
    while (off < n) {
        int w;
        if (poll_until(u, POLLOUT, deadline) < 0) return -1;
        w = (int)send(u, line + off, (size_t)(n - off), MSG_NOSIGNAL);
        if (w < 0) {
            if (errno == EINTR || errno == EAGAIN || errno == EWOULDBLOCK) continue;
            return -1;
        }
        if (w == 0) {
            errno = ECONNRESET;
            return -1;
        }
        off += w;
    }
    rn = 0;
    while (rn < (int)sizeof resp - 1) {
        char c;
        int r;
        if (poll_until(u, POLLIN, deadline) < 0) return -1;
        r = (int)recv(u, &c, 1, 0);
        if (r < 0) {
            if (errno == EINTR || errno == EAGAIN || errno == EWOULDBLOCK) continue;
            return -1;
        }
        if (r == 0) break;
        resp[rn++] = c;
        if (c == '\n') break;
    }
    resp[rn] = 0;
    if (strncmp(resp, "OK\n", 3) == 0) return 1;
    errno = EACCES;
    return 0;
}

static void copy_loop(int from, int to) {
    char buf[8192];
    for (;;) {
        ssize_t n = read(from, buf, sizeof buf);
        ssize_t off = 0;
        int bad = 0;
        if (n < 0) {
            if (errno == EINTR) continue;
            break;
        }
        if (n == 0) break;
        while (off < n) {
            ssize_t w = send(to, buf + off, (size_t)(n - off), MSG_NOSIGNAL);
            if (w < 0) {
                if (errno == EINTR) continue;
                bad = 1;
                break;
            }
            if (w == 0) {
                bad = 1;
                break;
            }
            off += w;
        }
        if (bad) break;
    }
    shutdown(to, SHUT_WR);
}

static void *copy_start(void *arg) {
    struct copy_arg *a = arg;
    copy_loop(a->from, a->to);
    return NULL;
}

static int discard_n(int fd, int n) {
    char buf[8192];
    while (n > 0) {
        int chunk = n > (int)sizeof buf ? (int)sizeof buf : n;
        ssize_t r = read(fd, buf, (size_t)chunk);
        if (r < 0) {
            if (errno == EINTR) continue;
            return -1;
        }
        if (r == 0) return -1;
        n -= (int)r;
    }
    return 0;
}

static int fill_until_block(int fd, int *pad) {
    char buf[8192];
    int total = 0;
    memset(buf, 0, sizeof buf);
    for (;;) {
        ssize_t w = send(fd, buf, sizeof buf, MSG_NOSIGNAL);
        if (w < 0) {
            if (errno == EINTR) continue;
            if (errno == EAGAIN || errno == EWOULDBLOCK) {
                *pad = total;
                return total > 0 ? 0 : -1;
            }
            return -1;
        }
        if (w == 0) return -1;
        total += (int)w;
        if (total > 8 * 1024 * 1024) {
            errno = ENOMEM;
            return -1;
        }
    }
}

/* Hold the caller's end full until the handshake finishes. A normal
   connected socketpair is writable immediately, which would make curl
   treat connect as done and then block in recv if O_NONBLOCK was lost.
   Filling the buffer keeps poll() waiting, so --max-time can fire. */
static void *proxy_thread(void *arg) {
    struct proxy_job *job = arg;
    long deadline = mono_ms() + HANDSHAKE_MS;
    int u = dial_unix_nb(job->path, deadline);
    int hs = -1;
    int fl;
    pthread_t th;
    struct copy_arg one;

    if (u >= 0) hs = proxy_handshake(u, job->host, job->port, deadline);
    if (hs != 1) {
        shutdown(job->client, SHUT_RDWR);
        close(job->client);
        if (u >= 0) close(u);
        free(job);
        return NULL;
    }
    if (discard_n(job->client, job->pad) < 0) {
        shutdown(job->client, SHUT_RDWR);
        close(job->client);
        close(u);
        free(job);
        return NULL;
    }
    fl = fcntl(u, F_GETFL);
    if (fl >= 0) fcntl(u, F_SETFL, fl & ~O_NONBLOCK);
    one.from = job->client;
    one.to = u;
    if (pthread_create(&th, NULL, copy_start, &one) != 0) {
        shutdown(job->client, SHUT_RDWR);
        close(job->client);
        close(u);
        free(job);
        return NULL;
    }
    copy_loop(u, job->client);
    pthread_join(th, NULL);
    close(u);
    close(job->client);
    free(job);
    return NULL;
}

static int proxy_connect_block(int fd, const char *host, int port, const char *path, int fl, int fdfl) {
    long deadline = mono_ms() + HANDSHAKE_MS;
    int u = dial_unix_nb(path, deadline);
    int hs, ufl;
    if (u < 0) return -1;
    hs = proxy_handshake(u, host, port, deadline);
    if (hs != 1) {
        int e = errno;
        close(u);
        errno = e ? e : EACCES;
        return -1;
    }
    ufl = fcntl(u, F_GETFL);
    if (ufl >= 0) fcntl(u, F_SETFL, ufl & ~O_NONBLOCK);
    if (dup2(u, fd) < 0) {
        int e = errno;
        close(u);
        errno = e;
        return -1;
    }
    if (u != fd) close(u);
    if (apply_status(fd, fl) < 0 || apply_fdflags(fd, fdfl) < 0) {
        errno = EIO;
        return -1;
    }
    return 0;
}

static int proxy_connect_nb(int fd, const char *host, int port, const char *path, int fl, int fdfl) {
    int sv[2];
    int pad = 0;
    struct proxy_job *job;
    pthread_t th;
    pthread_attr_t attr;
    int pr;

    if (socketpair(AF_UNIX, SOCK_STREAM, 0, sv) < 0) return -1;
    if (fcntl(sv[1], F_SETFD, FD_CLOEXEC) < 0 || apply_status(sv[0], fl | O_NONBLOCK) < 0) {
        close(sv[0]);
        close(sv[1]);
        return -1;
    }
    if (fill_until_block(sv[0], &pad) < 0) {
        int e = errno;
        close(sv[0]);
        close(sv[1]);
        errno = e;
        return -1;
    }
    job = calloc(1, sizeof *job);
    if (!job) {
        close(sv[0]);
        close(sv[1]);
        errno = ENOMEM;
        return -1;
    }
    job->client = sv[1];
    job->pad = pad;
    job->port = port;
    snprintf(job->path, sizeof job->path, "%s", path);
    snprintf(job->host, sizeof job->host, "%s", host);
    pthread_attr_init(&attr);
    pthread_attr_setdetachstate(&attr, PTHREAD_CREATE_DETACHED);
    pr = pthread_create(&th, &attr, proxy_thread, job);
    pthread_attr_destroy(&attr);
    if (pr != 0) {
        close(sv[0]);
        close(sv[1]);
        free(job);
        errno = EAGAIN;
        return -1;
    }
    if (dup2(sv[0], fd) < 0) {
        int e = errno;
        close(sv[0]);
        errno = e;
        return -1;
    }
    close(sv[0]);
    apply_fdflags(fd, fdfl);
    errno = EINPROGRESS;
    return -1;
}

static int proxy_connect(int fd, const char *host, int port) {
    const char *path;
    int fl, fdfl;

    path = getenv("LGB_PROXY_SOCK");
    if (!path || !path[0] || !real_connect) {
        errno = EACCES;
        return -1;
    }
    {
        struct sockaddr_un lim;
        if (strlen(path) >= sizeof lim.sun_path) {
            errno = ENAMETOOLONG;
            return -1;
        }
    }
    fl = fcntl(fd, F_GETFL);
    fdfl = fcntl(fd, F_GETFD);
    if (fl < 0 || fdfl < 0) return -1;
    /* dup2 of a fresh socket drops O_NONBLOCK and FD_CLOEXEC. Put the
       caller's flags back. A non-blocking caller cannot sit inside the
       handshake: curl's --max-time runs in its poll loop, which does not
       run while connect() is blocked in the daemon dial. */
    if (fl & O_NONBLOCK) return proxy_connect_nb(fd, host, port, path, fl, fdfl);
    return proxy_connect_block(fd, host, port, path, fl, fdfl);
}

int connect(int fd, const struct sockaddr *addr, socklen_t len) {
    pthread_once(&once, init_syms);
    if (!real_connect) {
        errno = EACCES;
        return -1;
    }
    if (!addr || !getenv("LGB_PROXY_SOCK")) {
        return real_connect(fd, addr, len);
    }
    if (addr->sa_family == AF_INET) {
        const struct sockaddr_in *in = (const struct sockaddr_in *)addr;
        uint32_t ip = ntohl(in->sin_addr.s_addr);
        char host[256];
        int port = ntohs(in->sin_port);
        if ((ip >> 24) == 127) {
            return real_connect(fd, addr, len);
        }
        if ((ip >> 24) == 0xf0) {
            int idx = (int)(ip & 0xff);
            pthread_mutex_lock(&mu);
            if (idx < 1 || idx > hostn || !hosts[idx][0]) {
                pthread_mutex_unlock(&mu);
                errno = EACCES;
                return -1;
            }
            snprintf(host, sizeof host, "%s", hosts[idx]);
            pthread_mutex_unlock(&mu);
        } else if (!inet_ntop(AF_INET, &in->sin_addr, host, sizeof host)) {
            errno = EACCES;
            return -1;
        }
        return proxy_connect(fd, host, port);
    }
    if (addr->sa_family == AF_INET6) {
        const struct sockaddr_in6 *in6 = (const struct sockaddr_in6 *)addr;
        char host[128];
        int port = ntohs(in6->sin6_port);
        if (IN6_IS_ADDR_LOOPBACK(&in6->sin6_addr)) {
            return real_connect(fd, addr, len);
        }
        if (!inet_ntop(AF_INET6, &in6->sin6_addr, host, sizeof host)) {
            errno = EACCES;
            return -1;
        }
        return proxy_connect(fd, host, port);
    }
    return real_connect(fd, addr, len);
}

/* The jail has no /etc/passwd. cat of that file must keep failing.
   ssh still calls getpwuid and exits if it gets nothing, so answer with a
   synthetic user that has no credentials and no host home. */

static int fill_pw(struct passwd *pwd, char *buf, size_t buflen, uid_t uid) {
    const char *fields[] = {"bot", "x", "bot", "/work", "/bin/sh"};
    size_t need = 0;
    int i;
    char *p;
    char *out[5];
    for (i = 0; i < 5; i++) need += strlen(fields[i]) + 1;
    if (buflen < need) return ERANGE;
    p = buf;
    for (i = 0; i < 5; i++) {
        size_t n = strlen(fields[i]);
        out[i] = p;
        memcpy(p, fields[i], n + 1);
        p += n + 1;
    }
    pwd->pw_name = out[0];
    pwd->pw_passwd = out[1];
    pwd->pw_uid = uid;
    pwd->pw_gid = getgid();
    pwd->pw_gecos = out[2];
    pwd->pw_dir = out[3];
    pwd->pw_shell = out[4];
    return 0;
}

struct passwd *getpwuid(uid_t uid) {
    static char buf[64];
    static struct passwd pw;
    if (fill_pw(&pw, buf, sizeof buf, uid) != 0) return NULL;
    return &pw;
}

int getpwuid_r(uid_t uid, struct passwd *pwd, char *buf, size_t buflen, struct passwd **result) {
    int rc;
    if (!pwd || !result) return EINVAL;
    rc = fill_pw(pwd, buf, buflen, uid);
    if (rc != 0) {
        if (result) *result = NULL;
        return rc;
    }
    *result = pwd;
    return 0;
}

struct passwd *getpwnam(const char *name) {
    if (!name || strcmp(name, "bot") != 0) return NULL;
    return getpwuid(getuid());
}

int getpwnam_r(const char *name, struct passwd *pwd, char *buf, size_t buflen, struct passwd **result) {
    if (!name || strcmp(name, "bot") != 0) {
        if (result) *result = NULL;
        return 0;
    }
    return getpwuid_r(getuid(), pwd, buf, buflen, result);
}

int getaddrinfo(const char *node, const char *service,
                const struct addrinfo *hints, struct addrinfo **res) {
    struct addrinfo *tmp = NULL;
    struct addrinfo hint;
    const struct addrinfo *hp = hints;
    int port = 0;
    int idx;
    int rc;
    struct addrinfo *ai;
    struct sockaddr_in *sa;

    pthread_once(&once, init_syms);
    if (!real_getaddrinfo) return EAI_FAIL;
    if (!node || !node[0] || strcmp(node, "localhost") == 0 || !getenv("LGB_PROXY_SOCK")) {
        return real_getaddrinfo(node, service, hints, res);
    }
    {
        unsigned char buf[16];
        if (inet_pton(AF_INET, node, buf) == 1 || inet_pton(AF_INET6, node, buf) == 1) {
            return real_getaddrinfo(node, service, hints, res);
        }
    }
    if (hints && (hints->ai_flags & AI_NUMERICHOST)) {
        return real_getaddrinfo(node, service, hints, res);
    }
    if (hints) {
        hint = *hints;
        hint.ai_family = AF_INET;
        hp = &hint;
    }
    rc = real_getaddrinfo("127.0.0.1", service, hp, &tmp);
    if (rc == 0 && tmp && tmp->ai_addr && tmp->ai_addr->sa_family == AF_INET) {
        port = ((struct sockaddr_in *)tmp->ai_addr)->sin_port;
    }
    if (tmp) freeaddrinfo(tmp);
    if (rc != 0) {
        if (service && isdigit((unsigned char)service[0])) {
            port = htons(atoi(service));
        } else if (service && service[0]) {
            return rc;
        }
    }
    pthread_mutex_lock(&mu);
    if (hostn >= 255) {
        pthread_mutex_unlock(&mu);
        return EAI_MEMORY;
    }
    idx = ++hostn;
    lower_copy(hosts[idx], sizeof hosts[idx], node);
    pthread_mutex_unlock(&mu);

    ai = calloc(1, sizeof *ai);
    sa = calloc(1, sizeof *sa);
    if (!ai || !sa) {
        free(ai);
        free(sa);
        return EAI_MEMORY;
    }
    sa->sin_family = AF_INET;
    sa->sin_port = (in_port_t)port;
    sa->sin_addr.s_addr = htonl(0xF0000000u | (unsigned)idx);
    ai->ai_family = AF_INET;
    ai->ai_socktype = (hints && hints->ai_socktype) ? hints->ai_socktype : SOCK_STREAM;
    ai->ai_protocol = (hints && hints->ai_protocol) ? hints->ai_protocol : IPPROTO_TCP;
    ai->ai_addrlen = sizeof *sa;
    ai->ai_addr = (struct sockaddr *)sa;
    *res = ai;
    return 0;
}
